# ==============================================================================
# ΡΥΘΜΙΣΕΙΣ INAV ΓΙΑ ΤΟ ΠΛΑΝΗΤΙΚΟ DRONE-CANSAT (GEPRC TAKER G4 35A AIO)
# Προσοχή: Πρώτα ρυθμίστε τις θύρες UART στην καρτέλα Ports του INAV Configurator
# ==============================================================================

# --- Ενεργοποίηση βασικών λειτουργιών ---
feature GPS                     # Ενεργοποίηση δέκτη GPS (MATEK M10Q-5883)
feature TELEMETRY               # Ενεργοποίηση αποστολής τηλεμετρίας στο έδαφος
feature AIRMODE                 # Κρατάει σταθερό το drone ακόμα και σε ελάχιστο γκάζι

# --- Ρυθμίσεις GPS & Πυξίδας ---
set gps_provider = UBLOX        # Πρωτόκολλο U-blox για το τσιπ M10
set gps_ublox_use_galileo = ON  # Ταυτόχρονη χρήση Ευρωπαϊκών δορυφόρων Galileo (ταχύτερο lock)
set mag_hardware = QMC5883      # Εξωτερική πυξίδα QMC5883L πάνω στο M10Q μέσω I2C

# --- Ασφάλεια 2 Σταδίων & Πτήση σε Ανώμαλο Έδαφος ---
set small_angle = 180           # Επιτρέπει όπλιση του Drone ακόμα κι αν πατάει στραβά σε κρατήρα (>25°)
set failsafe_procedure = RTH    # Αν χαθεί το σήμα ELRS στην πτήση Drone, γυρίζει πίσω μόνο του (ΟΧΙ Drop!)
set nav_rth_altitude = 1500     # Ύψος επιστροφής 1500 cm (= 15 μέτρα πάνω από εμπόδια/λακκούβες)
set nav_rth_climb_first = OFF   # Επιστρέφει κατευθείαν αν είναι ήδη ψηλά
set nav_rth_allow_landing = ALWAYS # Προσγειώνεται αυτόματα μόλις φτάσει στο σημείο απογείωσης

# --- Προστασία Μπαταρίας GNB 4S 650mAh 120C ---
set vbat_min_cell_voltage = 330     # Κρίσιμο όριο 3.30V ανά κυψέλη (13.2V σύνολο)
set vbat_warning_cell_voltage = 345 # Προειδοποίηση 3.45V ανά κυψέλη (13.8V σύνολο)
set vbat_max_cell_voltage = 425     # Μέγιστο όριο ανίχνευσης 4.25V

# --- Κινητήρες GR1404 4500KV & Bluejay ESCs ---
set motor_pwm_protocol = DSHOT600   # Ψηφιακή επικοινωνία DShot600 (μηδέν θόρυβος, χωρίς καλιμπράρισμα)
set dshot_beeper_enabled = ON       # Μετατρέπει τα μοτέρ σε ηχητικό beacon εύρεσης στο έδαφος!

# --- Όρια Κλίσης για Ασφαλή Πτήση Πάνω από Εμπόδια ---
set max_angle_inclination_rll = 350 # Μέγιστη πλάγια κλίση 35.0° σε Angle Mode
set max_angle_inclination_pit = 350 # Μέγιστη κλίση μπροστά/πίσω 35.0° σε Angle Mode

save                                # Αποθήκευση όλων των ρυθμίσεων στη μνήμη του STM32G473
